26
27/// Creates an HTTP router using the specified server configuration.
28pub fn create_router_with_config(config: &ServerConfig) -> Router {
29    let cache_manager = Arc::new(PagedKvCacheManager::new(
30        config.cache_total_blocks,
31        config.cache_block_size,
32    ));
33    let state = AppState {
34        cache_manager,
35        next_seq_id: Arc::new(AtomicU64::new(1)),
36        total_requests: Arc::new(AtomicU64::new(0)),
37        total_tokens_generated: Arc::new(AtomicU64::new(0)),
38    };
39
40    Router::new()
41        .route("/", get(index_handler))
42        .route("/health", get(health_handler))
43        .route("/metrics", get(metrics_handler))
44        .route("/v1/chat/completions", post(chat_completions_handler))
45        .with_state(state)
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51    use axum::body::Body;
52    use axum::http::{Request, StatusCode};
53    use tower::ServiceExt; // for `oneshot`
54
55    #[tokio::test]
56    async fn test_index_endpoint() {
57        let app = create_router();
58
59        let response = app
60            .oneshot(Request::builder().uri("/").body(Body::empty()).unwrap())
61            .await
62            .unwrap();
63
64        assert_eq!(response.status(), StatusCode::OK);
65        let body_bytes = axum::body::to_bytes(response.into_body(), usize::MAX)
66            .await
67            .unwrap();
68        let body_str = String::from_utf8(body_bytes.to_vec()).unwrap();
69        assert!(body_str.contains("spec-stream"));
70        assert!(body_str.contains("Speculative Decoding"));
71    }
